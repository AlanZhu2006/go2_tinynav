1. Minimal daily flow
===============================================================================

# Build a fresh map from RealSense.
# Walk slowly. Press Ctrl-C once to stop recording; the script then validates
# the bag and builds the map.
source /home/nvidia/twork/tinynav_setup.bash
cd /home/nvidia/twork/tinynav
bash scripts/tinynav_auto_map.sh

# Latest map symlink.
ls -lah /home/nvidia/twork/tinynav/output/latest_map

# Start navigation using output/latest_map.
# This starts RViz, TinyNav nodes, RViz-goal-to-POI bridge, diagnostics, and
# the Go2 bridge by default.
source /home/nvidia/twork/tinynav_setup.bash
cd /home/nvidia/twork/tinynav
bash scripts/tinynav_auto_nav.sh

# Start navigation without sending /cmd_vel to Go2.
bash scripts/tinynav_auto_nav.sh --no-go2

# Use a specific map.
bash scripts/tinynav_auto_nav.sh --map /home/nvidia/twork/tinynav/output/latest_map

# Attach / stop.
tmux attach -t tinynav_nav_auto
tmux kill-session -t tinynav_nav_auto

source /home/nvidia/twork/tinynav_setup.bash

# Device, firmware, USB speed.
rs-fw-update -l

  initial_reset:=true \
  tf_publish_rate:=1.0 \
  publish_tf:=true \
  enable_depth:=true \
  enable_color:=true \
  enable_infra1:=true \
  enable_infra2:=true \
  enable_gyro:=true \
  enable_accel:=true \
  unite_imu_method:=2 \
  depth_module.depth_profile:=848x480x30 \
  depth_module.infra_profile:=848x480x30 \
  rgb_camera.color_profile:=848x480x30

5. RViz goal to TinyNav POI
===============================================================================

# In RViz, use "2D Goal Pose". The bridge converts /goal_pose into:
#   <map>/pois.json
#   /mapping/cmd_pois
#   /rviz_goal_pose_marker
#   /rviz_goal_marker

source /home/nvidia/twork/tinynav_setup.bash
cd /home/nvidia/twork/tinynav
uv run python tool/rviz_goal_to_poi.py \
  --tinynav_map_path /home/nvidia/twork/tinynav/output/latest_map \
  --z 0.0 \
  --marker-z-offset 1.5

# Append instead of replacing the active single POI.
uv run python tool/rviz_goal_to_poi.py \
  --tinynav_map_path /home/nvidia/twork/tinynav/output/latest_map \
  --append \
  --z 0.0

# Inspect current POIs.
cat /home/nvidia/twork/tinynav/output/latest_map/pois.json
ros2 topic echo /mapping/cmd_pois

6. Localization and goal-success diagnostics
===============================================================================

source /home/nvidia/twork/tinynav_setup.bash

# Current pose in the saved map frame.
ros2 topic echo /mapping/current_pose_in_map

# Successful visual relocalization poses. If this is quiet, global relocalization
# is not currently succeeding.
ros2 topic echo /map/relocalization

# Navigation progress.
ros2 topic echo /mapping/nav_progress

# Goal success. This is not latched; start listening before a run.
ros2 topic echo /mapping/nav_done

# Local and global planning outputs.
ros2 topic echo /mapping/global_plan
ros2 topic echo /planning/trajectory_path
ros2 topic echo /cmd_vel

# Log checks.
tmux capture-pane -pt tinynav_nav_auto:map -S -5000 | rg 'nav done|All POIs|relocaliz|not enough|PnP'
tmux capture-pane -pt tinynav_nav_auto:go2-bridge -S -200 | rg 'Move|Released|StopMove'

8. VNC and RViz
===============================================================================

# Current stable headless display:
#   address: <jetson-lan-ip>:5901
#   password: set locally with TINYNAV_VNC_PASSWORD; do not commit it
#   display: :1
#   server: TigerVNC
#
# The real GNOME :0 desktop is black when HDMI is disconnected. Use :1.

export TINYNAV_VNC_PASSWORD='<private-vnc-password>'
bash /home/nvidia/twork/tinynav/scripts/run_visible_vnc.sh
vncserver -list
ss -ltnp | rg ':5901|Xtigervnc'

# Start RViz manually in VNC.
source /home/nvidia/twork/tinynav_setup.bash
cd /home/nvidia/twork/tinynav
export DISPLAY=:1
export XAUTHORITY=/home/nvidia/.Xauthority
export QT_X11_NO_MITSHM=1
export LIBGL_ALWAYS_SOFTWARE=1
ros2 run rviz2 rviz2 -d /tinynav/docs/vis_with_global_map.rviz

# Move RViz onscreen if openbox places it at the corner.
DISPLAY=:1 XAUTHORITY=/home/nvidia/.Xauthority wmctrl -r RViz -e 0,0,0,1280,720

# Screenshot from server side.
DISPLAY=:1 XAUTHORITY=/home/nvidia/.Xauthority scrot /tmp/vnc_rviz.png
file /tmp/vnc_rviz.png

9. Unitree Go2 bridge and keyboard test
===============================================================================

# Wired Go2 network. The static IP command is temporary and disappears after
# reboot. Do not run teleop and TinyNav navigation at the same time unless you
# intentionally want multiple /cmd_vel publishers.

sudo ip addr add 192.168.123.100/24 dev eth0 2>/dev/null || true
sudo ip link set eth0 up
ping -I eth0 -c 1 192.168.123.161

# Start Go2 bridge in foreground.
source /home/nvidia/twork/tinynav_setup.bash
cd /home/nvidia/twork/tinynav
export UNITREE_NET_IF=eth0
export GO2_CMD_TOPIC=/cmd_vel
export GO2_MAX_VX=0.45
export GO2_MAX_VY=0.25
export GO2_MAX_WZ=0.80
export GO2_MIN_CMD_V=0.10
export GO2_MIN_CMD_W=0.20
export GO2_LOG_COMMANDS=true
bash scripts/run_go2_cmd_bridge.sh

# Keyboard teleop in another terminal.
source /home/nvidia/twork/tinynav_setup.bash
cd /home/nvidia/twork/tinynav
bash scripts/run_keyboard_teleop.sh --topic /cmd_vel --linear-speed 0.30 --angular-speed 0.55

# Controls:
